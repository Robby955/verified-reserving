\chapter{Prior uncertainty in Bornhuetter-Ferguson}

Mack, \emph{The Prediction Error of Bornhuetter/Ferguson} \cite{Mack2008BF}, calls the selected pattern and
the a priori ultimate practically independent in Section~4, property (b), pages 94--95, then
uses the product-variance formula on pages 95--96. This chapter freezes the observed
chain-ladder pattern of Definition~\ref{def:bf} and isolates the exact prior contribution.
Independence below strengthens the paper's description.

\begin{definition}[Random BF reserve]\label{def:bfPriorRv}\lean{VerifiedReserving.RandomTriangle.bfOutstandingFractionRv, VerifiedReserving.RandomTriangle.bfReservePriorRv}\leanok
\uses{def:bf, def:obsSigma}
For observed triangle $X$, accident year $i$, and random a priori ultimate $U$,
\[
  A_i = 1 - \frac{1}{\operatorname{cdf}_i(X)},
  \qquad R^{BF,U}_i = A_i U.
\]
The coefficient $A_i$ is measurable for the observed-data sigma-algebra.
\end{definition}

\begin{definition}[Independent prior]\label{def:bfPriorIndep}\lean{VerifiedReserving.BFPriorIndependent}\leanok
\uses{def:bfPriorRv}
The sigma-algebra generated by $U$ is independent of the observed-data sigma-algebra.
The variance below is mathlib's standard $\operatorname{Var}(U)$, equal on a probability
space to $\int (U-\int U\,d\mu)^2d\mu$ under the stated measurability hypothesis.
\end{definition}

\begin{theorem}[Conditional mean of the random-prior BF reserve]\label{thm:bfPriorMean}\lean{VerifiedReserving.condExp_bfReservePriorRv}\leanok
\uses{def:bfPriorRv, def:bfPriorIndep}
On a probability space, if $U$ is strongly measurable and integrable, $A_iU$ is integrable,
and the prior is independent of the observed triangle, then
\[
  E[R^{BF,U}_i \mid \mathrm{obsSigma}] = A_i E[U]
\]
almost surely. This is the conditional form of the expectation calculation on Mack's page 95.
\end{theorem}
\begin{proof}\leanok Pull the observed coefficient out of the conditional expectation, then
condition the independent prior on the observed-data sigma-algebra. \end{proof}

\begin{theorem}[Conditional variance contributed by the prior]\label{thm:bfPriorVar}\lean{VerifiedReserving.condVar_bfReservePriorRv}\leanok
\uses{def:bfPriorRv, def:bfPriorIndep, thm:bfPriorMean}
If the prior and the resulting BF reserve are square-integrable, then
\[
  E\left[(R^{BF,U}_i - A_iE[U])^2 \mid \mathrm{obsSigma}\right]
    = A_i^2\operatorname{Var}(U)
\]
almost surely. This is the prior-uncertainty component of the product-variance formula on
Mack's pages 95--96; it does not state the other components of the paper's prediction-error
approximation.
\end{theorem}
\begin{proof}\leanok Rewrite the centred reserve as $A_i(U-E[U])$, pull $A_i^2$ out of the
conditional expectation, and use independence for the squared centred prior. \end{proof}
