% Blueprint for verified-reserving: Mack's chain-ladder model in Lean 4.
% Conventions: accident years i = 0,...,n-1 and development years k = 0,...,n-1,
% both zero-based; C_{i,k} is observed when i + k <= n-1.

\chapter{Introduction}

This blueprint tracks the formalization of Mack's distribution-free chain-ladder model
\cite{Mack1993} in Lean 4 on top of mathlib. Each statement below carries the name of its
Lean declaration; green nodes in the dependency graph are proved, the others are open work
listed in the roadmap. Statements are given in the paper's own terms, and where a published
formula is an estimator or an approximation rather than an identity, the text says so.

A run-off triangle with $n$ accident years and $n$ development years is a function
$C : \mathbb{N} \to \mathbb{N} \to \mathbb{R}$; $C_{i,k}$ is the cumulative claims of accident
year $i$ at development year $k$, observed when $i + k \le n-1$. The latest observed entry of
accident year $i$ is $C_{i,\,n-1-i}$.

\chapter{The chain-ladder estimators}

\section{Definitions}

\begin{definition}[Contributors]\label{def:contributors}\lean{VerifiedReserving.contributors}\leanok
The accident years contributing to the development factor $k \to k+1$ are
$i = 0, \dots, n-k-2$, those with $C_{i,k+1}$ observed.
\end{definition}

\begin{definition}[Column sums]\label{def:S}\lean{VerifiedReserving.S, VerifiedReserving.T}\leanok
\uses{def:contributors}
$S_k = \sum_{i=0}^{n-k-2} C_{i,k}$ and $T_k = \sum_{i=0}^{n-k-2} C_{i,k+1}$.
\end{definition}

\begin{definition}[Development factor]\label{def:fhat}\lean{VerifiedReserving.fhat}\leanok
\uses{def:S}
$\hat f_k = T_k / S_k$. (In Lean, division by zero is zero.)
\end{definition}

\begin{definition}[Individual factor]\label{def:F}\lean{VerifiedReserving.F}\leanok
$F_{i,k} = C_{i,k+1} / C_{i,k}$.
\end{definition}

\begin{definition}[Mack's variance estimator]\label{def:sigma2}\lean{VerifiedReserving.sigma2}\leanok
\uses{def:fhat, def:F, def:contributors}
$\hat\sigma_k^2 = \frac{1}{n-k-2}\sum_{i=0}^{n-k-2} C_{i,k}\,(F_{i,k} - \hat f_k)^2$.
The value for $k = n-2$ (a single contributor) is an extrapolation convention in
\cite{Mack1993}, not an estimator; no theorem here uses it.
\end{definition}

\begin{definition}[Projection]\label{def:Chat}\lean{VerifiedReserving.Chat}\leanok
\uses{def:fhat}
$\hat C_{i,k} = C_{i,n-1-i} \prod_{j=n-1-i}^{k-1} \hat f_j$ for $k \ge n-1-i$.
\end{definition}

\begin{definition}[Ultimate and reserve]\label{def:reserve}\lean{VerifiedReserving.ultimate, VerifiedReserving.reserve}\leanok
\uses{def:Chat}
$\hat C_{i,n-1}$ is the chain-ladder ultimate and $\hat R_i = \hat C_{i,n-1} - C_{i,n-1-i}$ the reserve.
\end{definition}

\begin{definition}[Mack's MSEP estimator]\label{def:msep}\lean{VerifiedReserving.msep}\leanok
\uses{def:reserve, def:sigma2, def:S}
Mack (1993), Theorem 3:
\[
\widehat{\mathrm{msep}}(\hat R_i)
 = \hat C_{i,n-1}^2 \sum_{k=n-1-i}^{n-2} \frac{\hat\sigma_k^2}{\hat f_k^2}
   \Big(\frac{1}{\hat C_{i,k}} + \frac{1}{S_k}\Big).
\]
\end{definition}

\section{Deterministic identities}

\begin{lemma}[Weighted-average form]\label{lem:weighted}\lean{VerifiedReserving.fhat_eq_weighted_average}\leanok
\uses{def:fhat, def:F}
If $C_{i,k} \neq 0$ for every contributor $i$, then
$\hat f_k = \sum_{i} \frac{C_{i,k}}{S_k} F_{i,k}$.
\end{lemma}
\begin{proof}\leanok Termwise: $\frac{C_{i,k}}{S_k}\cdot\frac{C_{i,k+1}}{C_{i,k}} = \frac{C_{i,k+1}}{S_k}$. \end{proof}

\begin{lemma}\label{lem:T}\lean{VerifiedReserving.T_eq_sum_weighted_F}\leanok
\uses{def:S, def:F}
If $C_{i,k} \neq 0$ for every contributor, $T_k = \sum_i C_{i,k} F_{i,k}$.
\end{lemma}
\begin{proof}\leanok Cancel the denominator termwise. \end{proof}

\begin{lemma}[Sum-of-squares decomposition]\label{lem:sq}\lean{VerifiedReserving.weighted_sq_dev}\leanok
\uses{lem:T}
For any centre $f$, with $C_{i,k} \neq 0$ for every contributor,
$\sum_i C_{i,k}(F_{i,k}-f)^2 = \sum_i C_{i,k}F_{i,k}^2 - 2 f T_k + f^2 S_k$.
\end{lemma}
\begin{proof}\leanok Expand the square and use Lemma~\ref{lem:T}. \end{proof}

\begin{lemma}[Collapse at the chain-ladder centre]\label{lem:sq_fhat}\lean{VerifiedReserving.weighted_sq_dev_at_fhat}\leanok
\uses{lem:sq, def:fhat}
If moreover $S_k \neq 0$, then
$\sum_i C_{i,k}(F_{i,k}-\hat f_k)^2 = \sum_i C_{i,k}F_{i,k}^2 - S_k \hat f_k^2$.
This is the identity behind the $n-k-2$ degrees of freedom in $\hat\sigma_k^2$.
\end{lemma}
\begin{proof}\leanok Put $f = \hat f_k = T_k/S_k$ in Lemma~\ref{lem:sq}. \end{proof}

\begin{lemma}[Projection step]\label{lem:Chat_succ}\lean{VerifiedReserving.Chat_succ}\leanok
\uses{def:Chat}
For $k \ge n-1-i$, $\hat C_{i,k+1} = \hat C_{i,k}\,\hat f_k$.
\end{lemma}
\begin{proof}\leanok Split off the top factor of the product. \end{proof}

\begin{lemma}\label{lem:Chat_diag}\lean{VerifiedReserving.Chat_diag}\leanok
\uses{def:Chat}
$\hat C_{i,n-1-i} = C_{i,n-1-i}$.
\end{lemma}
\begin{proof}\leanok Empty product. \end{proof}

\begin{lemma}\label{lem:oldest}\lean{VerifiedReserving.reserve_zero_of_oldest}\leanok
\uses{def:reserve}
The oldest accident year is fully developed: $\hat R_0 = 0$.
\end{lemma}
\begin{proof}\leanok Empty product again. \end{proof}

\chapter{Mack 1999 equals Mack 1993}

\begin{definition}[Mack's 1999 recursion]\label{def:se2rec}\lean{VerifiedReserving.se2rec}\leanok
\uses{def:Chat, def:sigma2, def:S, def:fhat}
For $\alpha = 1$ and unit weights \cite{Mack1999}, starting from $\mathrm{se}^2 = 0$ on the
latest diagonal $d = n-1-i$ and stepping $m$ times,
\[
\mathrm{se}^2_{m+1} = \hat C_{i,d+m}^2\Big(\frac{\hat\sigma_{d+m}^2}{\hat C_{i,d+m}} + \frac{\hat\sigma_{d+m}^2}{S_{d+m}}\Big) + \mathrm{se}^2_m\,\hat f_{d+m}^2 .
\]
\end{definition}

\begin{definition}\label{def:mackTerm}\lean{VerifiedReserving.mackTerm}\leanok
\uses{def:sigma2, def:fhat, def:Chat, def:S}
The summand of the closed form:
$\mathrm{term}_{i,k} = \frac{\hat\sigma_k^2}{\hat f_k^2}\big(\frac{1}{\hat C_{i,k}} + \frac{1}{S_k}\big)$.
\end{definition}

\begin{lemma}\label{lem:msep_sum}\lean{VerifiedReserving.msep_eq_sum_mackTerm}\leanok
\uses{def:msep, def:mackTerm}
$\widehat{\mathrm{msep}}(\hat R_i) = \hat C_{i,n-1}^2 \sum_{k=n-1-i}^{n-2} \mathrm{term}_{i,k}$.
\end{lemma}
\begin{proof}\leanok By definition. \end{proof}

\begin{lemma}[Truncated closed form]\label{lem:se2rec_closed}\lean{VerifiedReserving.se2rec_eq_closed}\leanok
\uses{def:se2rec, def:mackTerm, lem:Chat_succ}
If $\hat f_k \neq 0$ for $d \le k < d+m$, then
$\mathrm{se}^2_m = \hat C_{i,d+m}^2 \sum_{k=d}^{d+m-1} \mathrm{term}_{i,k}$.
\end{lemma}
\begin{proof}\leanok Induction on $m$ using Lemma~\ref{lem:Chat_succ} and $\hat f_{d+m}^2/\hat f_{d+m}^2 = 1$. \end{proof}

\begin{theorem}[Mack 1999 = Mack 1993]\label{thm:recursion}\lean{VerifiedReserving.se2rec_eq_msep}\leanok
\uses{lem:se2rec_closed, lem:msep_sum}
For $i \le n-1$ and $\hat f_k \neq 0$ along the row, $i$ steps of the 1999 recursion return
exactly Mack's 1993 estimator: $\mathrm{se}^2_i = \widehat{\mathrm{msep}}(\hat R_i)$.
\end{theorem}
\begin{proof}\leanok Lemma~\ref{lem:se2rec_closed} with $m = i$, since $d + i = n-1$. \end{proof}

\chapter{Bornhuetter-Ferguson on the chain-ladder pattern}

\begin{definition}[Cumulative development factor, BF reserve and ultimate]\label{def:bf}\lean{VerifiedReserving.cdf, VerifiedReserving.bfReserve, VerifiedReserving.bfUltimate}\leanok
\uses{def:fhat, def:Chat}
$\mathrm{cdf}_i = \prod_{k=d}^{n-2}\hat f_k$; for an a priori ultimate $U$,
$R^{BF}_i = U\,(1 - 1/\mathrm{cdf}_i)$ and $\hat C^{BF}_{i,n-1} = C_{i,d} + R^{BF}_i$.
\end{definition}

\begin{lemma}[Linearity]\label{lem:bf_linear}\lean{VerifiedReserving.bfReserve_smul, VerifiedReserving.bfReserve_add}\leanok
\uses{def:bf}
$R^{BF}_i(aU) = a\,R^{BF}_i(U)$ and $R^{BF}_i(U+V) = R^{BF}_i(U) + R^{BF}_i(V)$.
\end{lemma}
\begin{proof}\leanok Ring identities. \end{proof}

\begin{theorem}[BF with the chain-ladder prior is chain ladder]\label{thm:bf_cl}\lean{VerifiedReserving.bfUltimate_of_ultimate, VerifiedReserving.bfReserve_of_ultimate}\leanok
\uses{def:bf, def:reserve}
If $\mathrm{cdf}_i \neq 0$ and $U = \hat C_{i,n-1}$, then $\hat C^{BF}_{i,n-1} = \hat C_{i,n-1}$ and
$R^{BF}_i = \hat R_i$.
\end{theorem}
\begin{proof}\leanok $\hat C_{i,n-1} = C_{i,d}\,\mathrm{cdf}_i$, so $U(1 - 1/\mathrm{cdf}_i) = C_{i,d}\mathrm{cdf}_i - C_{i,d}$. \end{proof}

\begin{lemma}[The BF reserve is a fraction of the prior]\label{lem:bf_frac}\lean{VerifiedReserving.one_le_cdf, VerifiedReserving.bfReserve_nonneg, VerifiedReserving.bfReserve_le}\leanok
\uses{def:bf}
If every development factor along the row is at least $1$, then $\mathrm{cdf}_i \ge 1$ and, for
$U \ge 0$, $0 \le R^{BF}_i(U) \le U$.
\end{lemma}
\begin{proof}\leanok A product of numbers at least $1$ is at least $1$; then $0 \le 1 - 1/\mathrm{cdf}_i \le 1$. \end{proof}

\chapter{The stochastic layer}

\section{Model}

\begin{definition}[Random triangle]\label{def:RandomTriangle}\lean{VerifiedReserving.RandomTriangle}\leanok
A random run-off triangle on a measurable space $\Omega$ is a family of random variables
$C_{i,k} : \Omega \to \mathbb{R}$ together with a filtration $\mathcal{D}_k$ such that $C_{i,j}$
is $\mathcal{D}_k$-measurable whenever $j \le k$ ("everything observed up to development year $k$").
\end{definition}

\begin{definition}[Assumption (M1)]\label{def:Mack1}\lean{VerifiedReserving.Mack1}\leanok
\uses{def:RandomTriangle}
$E[C_{i,k+1} \mid \mathcal{D}_k] = f_k\, C_{i,k}$ almost surely, for all $i$ and $k$.
This is Mack's first assumption in the form conditioned on $\mathcal{D}_k$; Mack states it
conditioned on the row's own history and obtains this form from independence across accident
years (M2). That passage is Theorem~\ref{thm:passage}.
\end{definition}

\begin{definition}[Estimators as random variables]\label{def:rv}\lean{VerifiedReserving.RandomTriangle.Srv, VerifiedReserving.RandomTriangle.fhatRv}\leanok
\uses{def:RandomTriangle, def:S, def:fhat}
$S_k(\omega)$ and $\hat f_k(\omega)$ are the deterministic estimators applied to the triangle
observed at $\omega$.
\end{definition}

\begin{lemma}\label{lem:Srv_meas}\lean{VerifiedReserving.RandomTriangle.stronglyMeasurable_Srv}\leanok
\uses{def:rv}
$S_k$ is $\mathcal{D}_k$-measurable.
\end{lemma}
\begin{proof}\leanok A finite sum of $\mathcal{D}_k$-measurable functions. \end{proof}

\section{Mack's Theorem 2}

\begin{theorem}[Conditional unbiasedness of the development factor]\label{thm:unbiased}\lean{VerifiedReserving.condExp_fhatRv}\leanok
\uses{def:Mack1, def:rv, lem:Srv_meas}
Assume (M1), that $S_k \neq 0$ almost surely, that each $C_{i,k+1}$ and $\hat f_k$ are
integrable. Then $E[\hat f_k \mid \mathcal{D}_k] = f_k$ almost surely.
\end{theorem}
\begin{proof}\leanok
$\hat f_k = S_k^{-1} \sum_i C_{i,k+1}$. The factor $S_k^{-1}$ is $\mathcal{D}_k$-measurable and
comes out of the conditional expectation; by (M1) the conditional expectation of the sum is
$f_k \sum_i C_{i,k} = f_k S_k$; cancel $S_k$ where it is nonzero.
\end{proof}

\begin{lemma}\label{lem:fhat_meas}\lean{VerifiedReserving.RandomTriangle.stronglyMeasurable_fhatRv}\leanok
\uses{def:rv, lem:Srv_meas}
$\hat f_j$ is $\mathcal{D}_k$-measurable whenever $j + 1 \le k$.
\end{lemma}
\begin{proof}\leanok $S_j$ is $\mathcal{D}_j \subseteq \mathcal{D}_k$-measurable and each $C_{i,j+1}$ is $\mathcal{D}_k$-measurable. \end{proof}

\begin{theorem}[Uncorrelated development factors, conditional form]\label{thm:uncorrelated}\lean{VerifiedReserving.condExp_fhatRv_mul}\leanok
\uses{thm:unbiased, lem:fhat_meas}
Under (M1), on a finite measure space, for $j < k$ (with the integrability and $S \neq 0$
hypotheses of Theorem~\ref{thm:unbiased} for both columns and $\hat f_j \hat f_k$ integrable),
$E[\hat f_j \hat f_k \mid \mathcal{D}_j] = f_j f_k$ almost surely.
\end{theorem}
\begin{proof}\leanok
Condition on $\mathcal{D}_k$ first: $\hat f_j$ is $\mathcal{D}_k$-measurable and comes out, and
$E[\hat f_k \mid \mathcal{D}_k] = f_k$ by Theorem~\ref{thm:unbiased}. Then the tower property
down to $\mathcal{D}_j$ and Theorem~\ref{thm:unbiased} again for column $j$.
\end{proof}

\begin{theorem}[Mack 1993, Theorem 2]\label{thm:mack2}\lean{VerifiedReserving.integral_fhatRv_mul, VerifiedReserving.integral_fhatRv}\leanok
\uses{thm:uncorrelated, thm:unbiased}
On a probability space, under the same hypotheses, $E[\hat f_k] = f_k$ and
$E[\hat f_j \hat f_k] = f_j f_k$ for $j < k$: the chain-ladder factors are unbiased and
uncorrelated, which is what makes $\prod_k \hat f_k$ an unbiased estimator of $\prod_k f_k$.
\end{theorem}
\begin{proof}\leanok Integrate the conditional statements. \end{proof}

\section{Mack's Theorem 1}

\begin{definition}[Chain-ladder projection as a random variable]\label{def:ChatRv}\lean{VerifiedReserving.RandomTriangle.ChatRv}\leanok
\uses{def:rv}
$\hat C_{i,d+m} = C_{i,d}\prod_{k=d}^{d+m-1}\hat f_k$ with $d = n-1-i$.
\end{definition}

\begin{lemma}\label{lem:ChatRv_meas}\lean{VerifiedReserving.RandomTriangle.stronglyMeasurable_ChatRv}\leanok
\uses{def:ChatRv, lem:fhat_meas}
$\hat C_{i,d+m}$ is $\mathcal{D}_{d+m}$-measurable.
\end{lemma}
\begin{proof}\leanok Induction on $m$; each new factor $\hat f_{d+m}$ is $\mathcal{D}_{d+m+1}$-measurable. \end{proof}

\begin{lemma}[Iterating (M1)]\label{lem:C_succ}\lean{VerifiedReserving.condExp_C_succ, VerifiedReserving.condExp_C_of_Mack1}\leanok
\uses{def:Mack1}
For $d \le k$, $E[C_{i,k+1} \mid \mathcal{D}_d] = f_k\,E[C_{i,k} \mid \mathcal{D}_d]$, hence
$E[C_{i,d+m} \mid \mathcal{D}_d] = C_{i,d}\prod_{k=d}^{d+m-1} f_k$.
\end{lemma}
\begin{proof}\leanok Tower property through $\mathcal{D}_k$, then (M1); induction on $m$. \end{proof}

\begin{theorem}[Mack 1993, Theorem 1]\label{thm:mack1}\lean{VerifiedReserving.condExp_ChatRv, VerifiedReserving.condExp_ultimate_eq}\leanok
\uses{lem:ChatRv_meas, thm:unbiased, lem:C_succ}
Under (M1), with the partial products integrable and the column sums a.s. nonzero,
$E[\hat C_{i,d+m} \mid \mathcal{D}_d] = C_{i,d}\prod_{k=d}^{d+m-1} f_k$, and therefore
$E[\hat C_{i,n-1} \mid \mathcal{D}_d] = E[C_{i,n-1} \mid \mathcal{D}_d]$: the chain-ladder
ultimate is a conditionally unbiased estimator of the true ultimate claims.
\end{theorem}
\begin{proof}\leanok
Condition on $\mathcal{D}_{d+m}$: the partial product comes out and the last factor has
conditional expectation $f_{d+m}$ (Theorem~\ref{thm:unbiased}); tower down to $\mathcal{D}_d$
and induct. Compare with Lemma~\ref{lem:C_succ}.
\end{proof}

\section{The variance assumptions and the estimation variance of $\hat f_k$}

\begin{definition}[Residuals, (M3) and (M2')]\label{def:Mack3}\lean{VerifiedReserving.RandomTriangle.eps, VerifiedReserving.Mack3, VerifiedReserving.Mack2'}\leanok
\uses{def:RandomTriangle}
With $\varepsilon_{i,k} = C_{i,k+1} - f_k C_{i,k}$: (M3) $E[\varepsilon_{i,k}^2 \mid \mathcal{D}_k] = \sigma_k^2 C_{i,k}$,
and (M2') $E[\varepsilon_{i,k}\varepsilon_{j,k} \mid \mathcal{D}_k] = 0$ for $i \neq j$, the
conditional uncorrelatedness across accident years that independence of accident years supplies.
\end{definition}

\begin{lemma}\label{lem:fhat_sub}\lean{VerifiedReserving.RandomTriangle.fhatRv_sub_eq, VerifiedReserving.RandomTriangle.sq_fhatRv_sub}\leanok
\uses{def:rv, def:Mack3}
On $\{S_k \neq 0\}$, $\hat f_k - f_k = S_k^{-1}\sum_i \varepsilon_{i,k}$ and
$(\hat f_k - f_k)^2 = S_k^{-2}\sum_i\sum_j \varepsilon_{i,k}\varepsilon_{j,k}$.
\end{lemma}
\begin{proof}\leanok $T_k - f_k S_k = \sum_i \varepsilon_{i,k}$; expand the square of the sum. \end{proof}

\begin{theorem}[Estimation variance of the development factor]\label{thm:var_fhat}\lean{VerifiedReserving.condExp_sq_fhatRv_sub}\leanok
\uses{lem:fhat_sub, lem:Srv_meas}
Under (M3) and (M2'), with the residual products integrable, on $\{S_k \neq 0\}$,
$E[(\hat f_k - f_k)^2 \mid \mathcal{D}_k] = \sigma_k^2 / S_k$
(Mack 1993, proof of Theorem 3; Mack 1999, $\mathrm{s.e.}(\hat f_k)^2 = \hat\sigma_k^2/S_k$).
\end{theorem}
\begin{proof}\leanok
$S_k^{-2}$ is $\mathcal{D}_k$-measurable and comes out; conditional expectation is linear over the
double sum; cross terms vanish by (M2'); diagonal terms give $\sigma_k^2\sum_i C_{i,k} = \sigma_k^2 S_k$.
\end{proof}

\section{Unbiasedness of the variance estimator}

\begin{lemma}[Sum of squares around $\hat f_k$ via residuals]\label{lem:sq_eps}\lean{VerifiedReserving.weighted_sq_dev_eps}\leanok
\uses{lem:sq, def:fhat}
With all contributing $C_{i,k} \neq 0$ and $S_k \neq 0$, for any $f$,
$\sum_i C_{i,k}(F_{i,k} - \hat f_k)^2 = \sum_i \varepsilon_{i}^2 / C_{i,k} - S_k(\hat f_k - f)^2$
where $\varepsilon_i = C_{i,k+1} - f C_{i,k}$.
\end{lemma}
\begin{proof}\leanok Two instances of Lemma~\ref{lem:sq} (centres $\hat f_k$ and $f$) and $T_k = \hat f_k S_k$. \end{proof}

\begin{definition}\label{def:sigma2Rv}\lean{VerifiedReserving.RandomTriangle.sigma2Rv, VerifiedReserving.RandomTriangle.wssRv}\leanok
\uses{def:sigma2, def:rv}
$\hat\sigma_k^2$ and the weighted sum of squares $W_k = \sum_i C_{i,k}(F_{i,k} - \hat f_k)^2$ as
random variables; $\hat\sigma_k^2 = W_k/(n-k-2)$.
\end{definition}

\begin{theorem}[Unbiasedness of the variance estimator]\label{thm:sigma2}\lean{VerifiedReserving.condExp_wssRv, VerifiedReserving.condExp_sigma2Rv}\leanok
\uses{def:sigma2Rv, lem:sq_eps, thm:var_fhat, def:Mack3}
Under (M3) and (M2'), with all contributing $C_{i,k}$ and $S_k$ a.s. nonzero and the stated
integrability, $E[W_k \mid \mathcal{D}_k] = (n-k-2)\,\sigma_k^2$ for $k+2 \le n$, and hence
$E[\hat\sigma_k^2 \mid \mathcal{D}_k] = \sigma_k^2$ for $k + 3 \le n$.
\end{theorem}
\begin{proof}\leanok
By Lemma~\ref{lem:sq_eps}, $W_k = \sum_i \varepsilon_i^2/C_{i,k} - S_k(\hat f_k - f_k)^2$.
Each $C_{i,k}^{-1}$ is $\mathcal{D}_k$-measurable, so (M3) gives $\sigma_k^2$ per term, $n-k-1$
terms in all; $S_k$ is $\mathcal{D}_k$-measurable and Theorem~\ref{thm:var_fhat} gives
$\sigma_k^2$ for the subtracted term.
\end{proof}

\section{Process variance and the exact MSEP (Theorem 3)}

\begin{lemma}[Conditional second moment]\label{lem:second_moment}\lean{VerifiedReserving.condExp_eps, VerifiedReserving.condExp_sq_C_succ, VerifiedReserving.condExp_sq_C_succ_tower}\leanok
\uses{def:Mack1, def:Mack3}
Under (M1) and (M3), $E[\varepsilon_{i,k} \mid \mathcal{D}_k] = 0$,
$E[C_{i,k+1}^2 \mid \mathcal{D}_k] = f_k^2 C_{i,k}^2 + \sigma_k^2 C_{i,k}$, and for $d \le k$,
$E[C_{i,k+1}^2 \mid \mathcal{D}_d] = f_k^2 E[C_{i,k}^2 \mid \mathcal{D}_d] + \sigma_k^2 E[C_{i,k} \mid \mathcal{D}_d]$.
\end{lemma}
\begin{proof}\leanok Expand $C_{i,k+1} = f_k C_{i,k} + \varepsilon_{i,k}$; the cross term vanishes; tower property. \end{proof}

\begin{definition}[Process variance]\label{def:procVar}\lean{VerifiedReserving.procVar, VerifiedReserving.procVar_eq_sum}\leanok
$V_0 = 0$, $V_{m+1} = f_{d+m}^2 V_m + \sigma_{d+m}^2\, c\prod_{l<m} f_{d+l}$; in closed form
$V_m = \sum_{j<m}\big(\prod_{l=j+1}^{m-1} f_{d+l}\big)^2 \sigma_{d+j}^2\, c \prod_{l<j} f_{d+l}$.
\end{definition}

\begin{theorem}[Process variance along a row]\label{thm:procVar}\lean{VerifiedReserving.condVar_C_eq_procVar}\leanok
\uses{lem:second_moment, lem:C_succ, def:procVar}
Under (M1) and (M3), $E[C_{i,d+m}^2 \mid \mathcal{D}_d] - E[C_{i,d+m} \mid \mathcal{D}_d]^2 = V_m$
with $c = C_{i,d}$.
\end{theorem}
\begin{proof}\leanok Induction on $m$ with Lemma~\ref{lem:second_moment} and Lemma~\ref{lem:C_succ}. \end{proof}

\begin{lemma}[Conditional MSEP decomposition]\label{lem:msep_decomp}\lean{VerifiedReserving.condExp_sq_sub_of_stronglyMeasurable}\leanok
For a $\mathcal{D}$-measurable predictor $P$ of $Y$,
$E[(P - Y)^2 \mid \mathcal{D}] = \big(E[Y^2 \mid \mathcal{D}] - E[Y \mid \mathcal{D}]^2\big) + (P - E[Y \mid \mathcal{D}])^2$.
\end{lemma}
\begin{proof}\leanok Expand the square; $P$ comes out of the conditional expectation. \end{proof}

\begin{theorem}[Mack's Theorem 3, exact form]\label{thm:msep}\lean{VerifiedReserving.condMsep_eq}\leanok
\uses{lem:msep_decomp, thm:procVar, lem:ChatRv_meas, thm:mack1}
Let $\mathcal{D}$ be the observed data. Assume $\hat C_{i,d+m}$ is $\mathcal{D}$-measurable and
that $\mathcal{D}$ adds nothing to $\mathcal{D}_d$ about row $i$'s future (the conditional mean and
second moment of $C_{i,d+m}$ given $\mathcal{D}$ agree with those given $\mathcal{D}_d$; this is
what independence across accident years supplies). Then under (M1) and (M3)
\[
E\big[(\hat C_{i,d+m} - C_{i,d+m})^2 \mid \mathcal{D}\big] = V_m + \big(\hat C_{i,d+m} - C_{i,d}\textstyle\prod_{l<m} f_{d+l}\big)^2 .
\]
\end{theorem}
\begin{proof}\leanok Lemma~\ref{lem:msep_decomp} with $P = \hat C_{i,d+m}$, then Theorem~\ref{thm:procVar} and Lemma~\ref{lem:C_succ}. \end{proof}

\begin{definition}[Mack's estimators of the two terms]\label{def:mackEst}\lean{VerifiedReserving.mackProcess, VerifiedReserving.mackEstimation}\leanok
\uses{def:procVar, def:sigma2, def:fhat}
$\widehat{V}$ is $V_m$ with $\hat f, \hat\sigma^2$ plugged in. The estimation-error term
$\hat C_{i,n-1}^2 \sum_k \hat\sigma_k^2/(\hat f_k^2 S_k)$ is obtained from $(\hat C - M)^2$ by
Mack's conditional-resampling step: replace $(\hat f_k - f_k)^2$ by $\sigma_k^2/S_k$
(Theorem~\ref{thm:var_fhat}), drop the cross terms (Theorem~\ref{thm:uncorrelated}), and plug in
$\hat f, \hat\sigma^2$. This is a definition, not a theorem: it is the approximation in Mack
(1993), and the exact conditional MSEP above is the object it approximates (Buchwalder,
Bühlmann, Merz and Wüthrich 2006 estimate the same object differently).
\end{definition}

\section{From independence across accident years to the $\mathcal{D}_k$ form}

Mack states (M1) and (M3) conditioned on one accident year's own history, and assumes separately
that the accident years are independent (M2). Every theorem above conditions on $\mathcal{D}_k$
instead. This section proves the passage.

\begin{definition}[Row $\sigma$-algebras]\label{def:rowSigma}\lean{VerifiedReserving.RandomTriangle.rowSigma, VerifiedReserving.RandomTriangle.rowSigmaAll, VerifiedReserving.RandomTriangle.otherRowsSigma, VerifiedReserving.RandomTriangle.otherRowsAll}\leanok
\uses{def:RandomTriangle}
$\sigma(C_{i,0},\dots,C_{i,k})$, the history of accident year $i$ up to development year $k$;
$\sigma(C_{i,j} : j)$, the whole of accident year $i$; and the joins of these over the other
accident years $i' \neq i$.
\end{definition}

\begin{definition}[(M2) and the row-conditioned assumptions]\label{def:rowAssumptions}\lean{VerifiedReserving.RowsIndep, VerifiedReserving.RowsGenerateD, VerifiedReserving.Mack1Row, VerifiedReserving.Mack3Row}\leanok
\uses{def:rowSigma}
(M2) is \texttt{RowsIndep}: the $\sigma$-algebras of the individual accident years are independent.
\texttt{RowsGenerateD} says that $\mathcal{D}_k$ is what its name means, the join of the rows' histories
up to development year $k$. (M1row) and (M3row) are Mack's assumptions conditioned on
$\sigma(C_{i,0},\dots,C_{i,k})$ rather than on $\mathcal{D}_k$.
\end{definition}

\begin{lemma}[Rectangles]\label{lem:rect}\lean{VerifiedReserving.setIntegral_inter_of_indep}\leanok
If $m_1'$ and $m_2$ are independent, $F$ is $m_1'$-measurable and integrable, $a \in m_1'$ and
$b \in m_2$, then $\int_{a \cap b} F = \mu(b) \int_a F$.
\end{lemma}
\begin{proof}\leanok Apply mathlib's \texttt{condExp\_indep\_eq} to the indicator $1_a F$, which is
$m_1'$-measurable, and integrate the resulting constant over $b$. \end{proof}

\begin{theorem}[Conditioning on an independent enlargement changes nothing]\label{thm:condExp_sup}\lean{VerifiedReserving.condExp_sup_of_indep}\leanok
\uses{lem:rect}
Let $m_1 \le m_1'$, let $f$ be $m_1'$-measurable and integrable, and let $m_2$ be independent of
$m_1'$. Then $E[f \mid m_1 \vee m_2] = E[f \mid m_1]$ almost surely.
\end{theorem}
\begin{proof}\leanok The sets $a \cap b$ with $a \in m_1$, $b \in m_2$ form a $\pi$-system
generating $m_1 \vee m_2$. On such a set both $\int f$ and $\int E[f \mid m_1]$ factor by
Lemma~\ref{lem:rect} into $\mu(b)$ times the integral over $a$, and those agree by the defining
property of $E[f \mid m_1]$. The class of sets on which the two set integrals agree is closed
under complements and countable disjoint unions, so it is all of $m_1 \vee m_2$; conclude by the
uniqueness of the conditional expectation. Mathlib has the case $m_1 = \bot$ only. \end{proof}

\begin{lemma}[One row against the others]\label{lem:indep_rows}\lean{VerifiedReserving.indep_rowSigmaAll_otherRowsAll, VerifiedReserving.indep_rowSigmaAll_otherRowsSigma, VerifiedReserving.RandomTriangle.D_eq_sup}\leanok
\uses{def:rowAssumptions}
Under (M2), accident year $i$ is independent of all the other accident years taken together, and
so of their observed histories. Under \texttt{RowsGenerateD}, $\mathcal{D}_k$ is the join of row $i$'s
history up to $k$ with the other rows' histories up to $k$.
\end{lemma}
\begin{proof}\leanok \texttt{indep\_iSup\_of\_disjoint} for $\{i\}$ against its complement, then
monotonicity of independence; the splitting of $\mathcal{D}_k$ is \texttt{iSup\_split\_single}. \end{proof}

\begin{theorem}[(M1) and (M3) in the $\mathcal{D}_k$ form]\label{thm:passage}\lean{VerifiedReserving.mack1_of_mack1Row, VerifiedReserving.mack3_of_mack3Row}\leanok
\uses{thm:condExp_sup, lem:indep_rows, def:Mack1, def:Mack3}
Assume (M2), that $\mathcal{D}_k$ is generated by the rows' histories, and the integrability of the
entries. Then Mack's row-conditioned (M1row) implies (M1), and (M3row) implies (M3).
\end{theorem}
\begin{proof}\leanok Split $\mathcal{D}_k$ by Lemma~\ref{lem:indep_rows}, apply
Theorem~\ref{thm:condExp_sup} with $m_1$ the row's history up to $k$, $m_1'$ the whole row and
$m_2$ the other rows, and finish with the row-conditioned assumption. \end{proof}

\begin{theorem}[(M2') is a consequence, not an assumption]\label{thm:mack2prime}\lean{VerifiedReserving.condExp_eps_rowSigma, VerifiedReserving.mack2'_of_rows}\leanok
\uses{thm:condExp_sup, lem:indep_rows, def:Mack3}
Under (M2), \texttt{RowsGenerateD} and (M1row), the residuals of two different accident years are
conditionally uncorrelated given $\mathcal{D}_k$: assumption (M2'), which the variance theorems
take as a hypothesis, follows from independence across accident years.
\end{theorem}
\begin{proof}\leanok Condition on $\mathcal{D}_k$ together with the whole of accident year $j$:
row $j$'s residual comes out of the conditional expectation, and what is left is row $i$'s
residual conditioned on a $\sigma$-algebra that adds nothing about row $i$, so it equals
$E[\varepsilon_{i,k} \mid \sigma(C_{i,0},\dots,C_{i,k})] = 0$ by
Theorem~\ref{thm:condExp_sup} and (M1row). The tower property brings the identity back to
$\mathcal{D}_k$. \end{proof}

\chapter{The variant catalogue: Mack versus conditional resampling}

Write $a_k = \hat\sigma_k^2/(\hat f_k^2 S_k)$ for the relative estimation variance of
development factor $k$ along the row. Mack's estimation-error term is
$\hat C_{i,n-1}^2\sum_k a_k$; the conditional-resampling term of Buchwalder, Bühlmann,
Merz and Wüthrich (2006), which Murphy (1994) also reaches, is
$\hat C_{i,n-1}^2\big(\prod_k (1+a_k) - 1\big)$.

\begin{definition}\label{def:bbmw}\lean{VerifiedReserving.relVar, VerifiedReserving.bbmwEstimation}\leanok
\uses{def:mackEst}
$a_k$ and the conditional-resampling estimator $\hat C_{i,n-1}^2(\prod_k (1+a_k) - 1)$.
\end{definition}

\begin{lemma}[Product minus sum]\label{lem:remainder}\lean{VerifiedReserving.one_add_sum_le_prod_one_add, VerifiedReserving.remainder_nonneg, VerifiedReserving.remainder_two, VerifiedReserving.remainder_eq_zero_of_subsingleton_support, VerifiedReserving.prod_one_add_le_exp_sum}\leanok
For $a_k \ge 0$: $1 + \sum a_k \le \prod (1+a_k) \le \exp(\sum a_k)$; for two terms
$(1+a)(1+b) - 1 - (a+b) = ab$; and $\prod(1+a_k) - 1 - \sum a_k = 0$ when at most one $a_k$ is
nonzero.
\end{lemma}
\begin{proof}\leanok Induction over the index set for the Weierstrass inequality; $1 + x \le e^x$ termwise for the exponential bound. \end{proof}

\begin{theorem}[Catalogue row: Mack is the first-order part of BBMW]\label{thm:catalogue}\lean{VerifiedReserving.mackEstimation_le_bbmwEstimation, VerifiedReserving.bbmwEstimation_sub_mackEstimation, VerifiedReserving.bbmwEstimation_eq_mackEstimation_of_one_factor, VerifiedReserving.bbmwEstimation_sub_mackEstimation_le}\leanok
\uses{def:bbmw, lem:remainder}
With $a_k \ge 0$ along the row: (i) Mack $\le$ BBMW; (ii) the difference is exactly
$\hat C_{i,n-1}^2\big(\prod_k(1+a_k) - 1 - \sum_k a_k\big)$; (iii) the two coincide for the
second-latest accident year, which has a single development factor; (iv) the difference is at most
$\hat C_{i,n-1}^2\big(e^{\sum a_k} - 1 - \sum a_k\big)$, second order in the relative estimation
variances.
\end{theorem}
\begin{proof}\leanok Lemma~\ref{lem:remainder} scaled by $\hat C_{i,n-1}^2$. \end{proof}

\begin{theorem}[Strict counterexample]\label{thm:counterexample}\lean{VerifiedReserving.Cex, VerifiedReserving.mackEstimation_lt_bbmwEstimation_Cex, VerifiedReserving.exists_mackEstimation_lt_bbmwEstimation}\leanok
\uses{thm:catalogue}
There is a run-off triangle (four accident years, four development years, with unequal individual
factors in the first two columns) and an accident year on which Mack's estimation-error term is
strictly smaller than the conditional-resampling term: the two are different estimators, not two
presentations of one. On the exhibited triangle $a_0 = 1/297$, $a_1 = 1/2809$, and the remainder
is $\hat C^2 a_0 a_1 > 0$. Every number is checked by \texttt{norm\_num} from the definitions.
\end{theorem}
\begin{proof}\leanok Evaluate the definitions on the triangle; apply Theorem~\ref{thm:catalogue}(ii). \end{proof}

\section{The total reserve: Mack's Corollary}

\begin{definition}[Total-reserve estimators]\label{def:total}\lean{VerifiedReserving.msepTotal, VerifiedReserving.mackCross, VerifiedReserving.bbmwCross, VerifiedReserving.mackTotalEstimation, VerifiedReserving.bbmwTotalEstimation}\leanok
\uses{def:msep, def:bbmw}
Mack's Corollary: $\widehat{\mathrm{msep}}(\hat R) = \sum_i \widehat{\mathrm{msep}}_i + \sum_i \hat C_i \big(\sum_{j>i}\hat C_j\big)\, 2\sum_{k \in \mathrm{row}\,i} a_k$,
the cross terms running over the older year's remaining factors. The conditional-resampling
aggregate replaces $2\sum_k a_k$ by $2(\prod_k (1+a_k) - 1)$ (Buchwalder et al. 2006, Section 4.3).
\end{definition}

\begin{theorem}[Aggregated catalogue row]\label{thm:total}\lean{VerifiedReserving.bbmwTotalEstimation_sub_mackTotalEstimation, VerifiedReserving.mackTotalEstimation_le_bbmwTotalEstimation}\leanok
\uses{def:total, lem:remainder}
The aggregated estimation-error terms differ by
$\sum_i \big(\hat C_i^2 + 2\hat C_i \sum_{j>i}\hat C_j\big)\big(\prod_{k \in \mathrm{row}\,i}(1+a_k) - 1 - \sum_{k \in \mathrm{row}\,i} a_k\big)$,
which is nonnegative when the ultimates and the $a_k$ are: Mack's total is at most the
resampling total, by the same row-wise second-order remainder as in the single-year case.
\end{theorem}
\begin{proof}\leanok Termwise algebra and Lemma~\ref{lem:remainder}. \end{proof}

\chapter{Non-vacuity: a nondegenerate model}

\begin{theorem}[A nondegenerate Mack model exists]\label{thm:nontrivial}\lean{VerifiedReserving.NontrivialModel.X, VerifiedReserving.NontrivialModel.exists_nontrivial_mack_model}\leanok
\uses{def:Mack1, def:Mack3}
There is a probability space, a random triangle with three accident years, and constants
$f_k$, $\sigma_k^2$ with $\sigma_0^2 > 0$, satisfying (M1), (M3) and (M2'), in which the first
development step is genuinely random. The model: eight equally likely outcomes, an independent
$\pm 1$ shock per accident year, $C_{i,0} = c$ and $C_{i,k} = c f^k (1 + s\,\xi_i)$ for $k \ge 1$,
with $\mathcal{D}_0$ trivial and $\mathcal{D}_k$ everything for $k \ge 1$.
\end{theorem}
\begin{proof}\leanok Conditional expectations given the trivial $\sigma$-algebra are eight-term averages; the shocks sum to zero, have unit square, and are orthogonal. \end{proof}

\begin{theorem}[Headline theorems instantiated]\label{thm:instantiated}\lean{VerifiedReserving.NontrivialModel.fhat0_unbiased, VerifiedReserving.NontrivialModel.ultimate_unbiased, VerifiedReserving.NontrivialModel.var_fhat0, VerifiedReserving.NontrivialModel.sigma2_unbiased}\leanok
\uses{thm:nontrivial, thm:unbiased, thm:mack1, thm:var_fhat, thm:sigma2}
On that model, $E[\hat f_0 \mid \mathcal{D}_0] = 2$ although $\hat f_0$ is random;
$E[\hat C_{2,2} \mid \mathcal{D}_0] = E[C_{2,2} \mid \mathcal{D}_0]$;
$E[(\hat f_0 - 2)^2 \mid \mathcal{D}_0] = \sigma_0^2 / S_0$; and $E[\hat\sigma_0^2 \mid \mathcal{D}_0] = \sigma_0^2 = 4$.
\end{theorem}
\begin{proof}\leanok Direct instantiation of the general theorems; every hypothesis is discharged. \end{proof}
