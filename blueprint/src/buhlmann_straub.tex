\chapter{Bühlmann-Straub credibility}

Bühlmann and Straub, \emph{Glaubwürdigkeit für Schadensätze} (1970), minimize expected squared
prediction error over unbiased linear estimators in Section~6, pages 118--120, and derive the
normal equations in display (1). Display (4), page 123, gives the exposure-weighted individual
mean, the credibility factor, and the credibility-weighted blend. The deterministic loss below
is an algebraic representation of the one-risk normal equation, not the expected
prediction-error objective of the paper.

\begin{definition}[Weighted experience, credibility, estimate, and loss]\label{def:bs}\lean{VerifiedReserving.buhlmannStraubTotalWeight, VerifiedReserving.buhlmannStraubWeightedTotal, VerifiedReserving.buhlmannStraubWeightedMean, VerifiedReserving.buhlmannStraubCredibility, VerifiedReserving.buhlmannStraubEstimate, VerifiedReserving.buhlmannStraubLoss}\leanok
For a finite sample with exposure weights $P_l$ and observations $X_l$, set
\[
  P_\bullet=\sum_l P_l,\qquad T=\sum_l P_lX_l,\qquad \bar X=T/P_\bullet,
  \qquad Z=\frac{P_\bullet w}{v+P_\bullet w}.
\]
For collective mean $m$, process variance $v$, and between-risk variance $w$, define
\[
  \widehat m=\frac{wT+vm}{v+P_\bullet w},\qquad
  L(t)=w\sum_lP_l(X_l-t)^2+v(t-m)^2.
\]
The quantities $m$, $v$, and $w$ are inputs.
\end{definition}

\begin{theorem}[Normal equation, square completion, and minimizer]\label{thm:bsMinimizer}\lean{VerifiedReserving.buhlmannStraub_weighted_sq_dev, VerifiedReserving.buhlmannStraubEstimate_normal_eq, VerifiedReserving.buhlmannStraubLoss_eq_at_estimate_add, VerifiedReserving.buhlmannStraubEstimate_minimizes}\leanok
\uses{def:bs}
If $v+P_\bullet w\ne0$, then
\[
  (v+P_\bullet w)\widehat m=wT+vm,
  \qquad L(t)=L(\widehat m)+(v+P_\bullet w)(t-\widehat m)^2.
\]
Consequently, $L(\widehat m)\le L(t)$ for every $t$ whenever $v+P_\bullet w>0$.
\end{theorem}
\begin{proof}\leanok Expand the weighted squared deviations, substitute the normal equation,
and use nonnegativity of a square. \end{proof}

\begin{theorem}[Credibility form, display (4)]\label{thm:bsCredibility}\lean{VerifiedReserving.buhlmannStraubEstimate_eq_credibility, VerifiedReserving.buhlmannStraubCredibility_mem_Icc}\leanok
\uses{def:bs, thm:bsMinimizer}
If $P_\bullet\ne0$ and $v+P_\bullet w\ne0$, then
\[
  \widehat m=Z\bar X+(1-Z)m.
\]
If $P_\bullet$, $v$, and $w$ are nonnegative and the denominator is positive, then
$0\le Z\le1$.
\end{theorem}
\begin{proof}\leanok Clear the two nonzero denominators and normalize the resulting polynomial
identity. The interval follows by comparing the nonnegative numerator with its denominator.
\end{proof}
